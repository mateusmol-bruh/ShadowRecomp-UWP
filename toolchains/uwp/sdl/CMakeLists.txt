cmake_minimum_required(VERSION 3.28)
project(ShadowUwpSdl LANGUAGES C CXX)
if(NOT WINDOWS_STORE OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    message(FATAL_ERROR "The SDL WinRT backend requires WindowsStore and MSVC for C++/CX.")
endif()
set(SDL_INSTALL ON CACHE BOOL "" FORCE)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
set(SDL_GPU OFF CACHE BOOL "" FORCE)
set(SDL_OPENGL OFF CACHE BOOL "" FORCE)
set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
set(SDL_HIDAPI OFF CACHE BOOL "" FORCE)
set(SDL_VIRTUAL_JOYSTICK ON CACHE BOOL "" FORCE)
set(CMAKE_CXX_STANDARD 17)
link_directories(BEFORE "$ENV{VCToolsInstallDir}/lib/x64/store")
add_compile_definitions(__XBOXSERIES__)
add_subdirectory("${SHADOW_SDL_UWP_SOURCE}" SDL3)
